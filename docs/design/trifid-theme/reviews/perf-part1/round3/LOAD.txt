== measure.sh, the script that ran the sequence (paths shortened: <repo> is this checkout, <scratch> a temporary folder; wt-prev is a checkout of 161d530b, wt-base of 6f10463e) ==
#!/bin/bash
# Round 3 measurements, one after another. Raw output to docs/.../perf-part1/round3/.
export PATH="$HOME/.nvm/versions/node/v22.23.3/bin:$PATH"
M=<repo>/frontcreck
SP=<scratch>
R=$M/../docs/design/trifid-theme/reviews/perf-part1/round3
S=$M/../docs/design/trifid-theme/reviews/perf-part1/scripts
mkdir -p $R
load1() { uptime | sed 's/.*load averages: //' | awk '{print $1}'; }
waitquiet() { # up to 30 minutes for the 1 minute load to fall under 4
  local t0=$(date +%s)
  while :; do
    local l=$(load1)
    if awk "BEGIN{exit !($l < 4)}"; then echo "waited $(( $(date +%s) - t0 )) s; load $l"; return 0; fi
    if [ $(( $(date +%s) - t0 )) -ge 1800 ]; then echo "waited 1800 s and the load never fell under 4; running at load $l"; return 1; fi
    sleep 20
  done
}
power() { pmset -g batt | head -1; }
for i in 1 2 3; do
  for v in current prev; do
    if [ $v = current ]; then D=$M; else D=$SP/wt-prev/frontcreck; fi
    f=$R/perf-$v-run$i.txt
    w=$(waitquiet)
    if ! power | grep -q "AC Power"; then echo "ON BATTERY: stopped before $v run $i" > $R/STOPPED-ON-BATTERY.txt; exit 1; fi
    cd $D
    { echo "# $(date '+%Y-%m-%d %H:%M:%S') $v ($(git rev-parse --short HEAD)); $w; load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; echo "> npm run perf"; npm run perf 2>&1; echo "# exit $?; load averages after: $(uptime | sed 's/.*load averages: //'); $(power)"; } > $f
    cp "$(ls -t scripts/perf/out/*.json | head -1)" $R/perf-$v-run$i.json
    echo "done $v $i: $(tail -1 $f)" >> $SP/measure.log
  done
done
cd $M
w=$(waitquiet)
{ echo "# $(date '+%Y-%m-%d %H:%M:%S') current; $w; load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; node ../docs/design/trifid-theme/reviews/baseline/hover-measure.mjs $R/hover.json http://127.0.0.1:3500 --start 2>&1; echo "# load averages after: $(uptime | sed 's/.*load averages: //')"; } > $R/hover.txt
echo "done hover" >> $SP/measure.log
$SP/servers.sh start >> $SP/measure.log
w=$(waitquiet)
{ echo "# $(date '+%Y-%m-%d %H:%M:%S'); $w; load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; node $S/startup-longtask.mjs $R/startup-longtask.json --rounds 6 baseline=http://127.0.0.1:3301 current=http://127.0.0.1:3302 prev=http://127.0.0.1:3303 2>&1; echo "# load averages after: $(uptime | sed 's/.*load averages: //')"; } > $R/startup-longtask.txt
{ echo "# $(date '+%Y-%m-%d %H:%M:%S') untimed WebGL calls; load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; node $S/startup-longtask.mjs $R/startup-longtask-plain.json --plain --rounds 5 baseline=http://127.0.0.1:3301 current=http://127.0.0.1:3302 2>&1; echo "# load averages after: $(uptime | sed 's/.*load averages: //')"; } > $R/startup-longtask-plain.txt
echo "done startup" >> $SP/measure.log
w=$(waitquiet)
{ echo "# $(date '+%Y-%m-%d %H:%M:%S'); $w; load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; node $S/software-frames.mjs $R/software-frames.json --rounds 3 baseline=http://127.0.0.1:3301 "full=http://127.0.0.1:3302?lite=off" lighter=http://127.0.0.1:3302 2>&1; echo "# load averages after: $(uptime | sed 's/.*load averages: //')"; } > $R/software-frames.txt
$SP/servers.sh stop >> $SP/measure.log
echo "done frames" >> $SP/measure.log
{ echo "# $(date '+%Y-%m-%d %H:%M:%S'); load averages before: $(uptime | sed 's/.*load averages: //'); $(power)"; node $S/fade-measure.mjs $R/fade-measure.json --runs 5 2>&1; echo "# load averages after: $(uptime | sed 's/.*load averages: //')"; } > $R/fade-measure.txt
echo "ALL DONE $(date)" >> $SP/measure.log
